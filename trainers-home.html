(() => { const user = typeof getCurrentUser === 'function' ? getCurrentUser() : null; const storageKey = `pebblio_trainer_${user?.userId || 'guest'}`; const readData = () => { try { return JSON.parse(localStorage.getItem(storageKey) || '{}'); } catch (error) { return {}; } }; const writeData = (next) => { try { localStorage.setItem(storageKey, JSON.stringify(next)); } catch (error) { console.warn('Trainer data could not be saved:', error); } }; const defaults = { streak: 12, xp: 820, accuracy: 88, minutes: 58, weeklyGoal: 72, focus: 'Algebra drill', completedToday: 3, recent: ['Algebra sprint', 'Reading comprehension', 'Vocabulary review'] }; const state = { ...defaults, ...readData() }; const widget = document.querySelector('[data-trainer-widget]'); const render = () => { if (!widget) return; widget.innerHTML = `
Practice Today
Streak
${state.streak}d
XP
${state.xp}
Weekly goal ${state.weeklyGoal}%
${state.focus}
${state.completedToday} sessions done
${state.accuracy}%
`; }; render(); window.pebblioTrainerWidget = { getState: () => readData(), update: (next) => { const merged = { ...state, ...next }; Object.assign(state, merged); writeData(state); render(); } }; })();
